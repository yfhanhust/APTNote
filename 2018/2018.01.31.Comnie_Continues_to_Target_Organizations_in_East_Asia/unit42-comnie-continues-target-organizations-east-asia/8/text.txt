import requests
import sys
import re
def decode(data):
o = ""
for c in data:
if c == "*":
o += "."
elif c == "|":
o += ":"
elif c == "+":
o += ";"
else:
o += chr(ord(c)-49)
return o
r = requests.get(sys.argv[1])
fd = r.text
data = fd.split("++a++")[1].split("++a++")[0]
print(decode(data))
