   def decrypt_cnc_msg(ct):
"""
Decrypt strings
   ct: bytearray of the ciphertext
   returns bytearray of the plaintext
"""
# seed
k = 0xff
for i in reversed(range(len(ct))):
 # XOR with previous
 k = ct[i] = ct[i] ^ k

return ct
